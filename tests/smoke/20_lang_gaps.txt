300 DEF PROCchg(RETURN p, RETURN q$, r%):p=p*10:q$=q$+"y":r%=99:ENDPROC
310 DEF FNtwice(RETURN v):v=v*2:=v+1
320 DEF PROCouter(RETURN o):PROCinner(o):ENDPROC
330 DEF PROCinner(RETURN i):i=i+100:ENDPROC
340 DEF PROCbuf(n%):LOCAL i%:DIM p% LOCAL n%:FOR i%=0 TO n%:p%?i%=i%:NEXT:PRINT "buf ";p%?n%;" ";(p%>&8000):ENDPROC
350 DEF FNsp:LOCAL s%:DIM s% LOCAL -1:=s%
